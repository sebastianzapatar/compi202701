2. Strings: de punta a punta

lexer/lexer.go
case '"':
    tok.Type = token.STRING
    tok.Literal = l.readString()

func (l *Lexer) readString() string {
    position := l.position + 1
    for {
        l.readChar()
        if l.ch == '"' || l.ch == 0 {
            break
        }
    }
    return l.input[position:l.position]
}

    Token token.Token
    Value string
}

object + evaluator
const STRING_OBJ = "STRING"

type String struct {
    Value string
}
func (s *String) Type() ObjectType { return STRING_OBJ }
func (s *String) Inspect() string  { return s.Value }

// Eval
case *ast.StringLiteral:
    return &object.String{Value: node.Value}

// evalInfixExpression, antes de los casos == / !=
case left.Type() == object.STRING_OBJ && right.Type() == object.STRING_OBJ:
    return evalStringInfixExpression(operator, left, right)

func evalStringInfixExpression(operator string, left, right object.Object) object.Object {
    if operator != "+" {
        return newError("unknown operator: %s %s %s",
            left.Type(), operator, right.Type())
    }
    leftVal := left.(*object.String).Value
    rightVal := right.(*object.String).Value
    return &object.String{Value: leftVal + rightVal}
}
Para pensar: "a" == "a" produce un error en Monkey. ¿Por qué no basta con el caso left == right de la Semana 6? (Pista: dos *object.String distintos tienen punteros distintos.)

3. Funciones integradas (built-ins)

Algunas cosas no se pueden escribir en Monkey mismo (¿cómo medirías un string sin ayuda?). Las funciones integradas son funciones Go expuestas a Monkey.

object/object.go + evaluator/builtins.go
type BuiltinFunction func(args ...Object) Object

type Builtin struct {
    Fn BuiltinFunction
}
func (b *Builtin) Type() ObjectType { return BUILTIN_OBJ }
func (b *Builtin) Inspect() string  { return "builtin function" }

// evaluator/builtins.go
var builtins = map[string]*object.Builtin{
    "len": &object.Builtin{
        Fn: func(args ...object.Object) object.Object {
            if len(args) != 1 {
                return newError("wrong number of arguments. got=%d, want=1",
                    len(args))
            }
            switch arg := args[0].(type) {
            case *object.String:
                return &object.Integer{Value: int64(len(arg.Value))}
            default:
                return newError("argument to `len` not supported, got %s",
                    args[0].Type())
            }
        },
    },
}
evaluator/evaluator.go
func evalIdentifier(node *ast.Identifier, env *object.Environment) object.Object {
    if val, ok := env.Get(node.Value); ok {
        return val
    }
    if builtin, ok := builtins[node.Value]; ok {
        return builtin
    }
    return newError("identifier not found: " + node.Value)
}

func applyFunction(fn object.Object, args []object.Object) object.Object {
    switch fn := fn.(type) {

    case *object.Function:
        extendedEnv := extendFunctionEnv(fn, args)
        evaluated := Eval(fn.Body, extendedEnv)
        return unwrapReturnValue(evaluated)

    case *object.Builtin:
        return fn.Fn(args...)

    default:
        return newError("not a function: %s", fn.Type())
    }
}
Orden de búsqueda: primero el entorno, luego los built-ins. Consecuencia: un usuario puede sombrear len con let len = fn(x) { 0 };. ¿Es buena idea? Es una decisión de diseño del lenguaje.

4. Arrays: literales y expresiones de índice

parser/parser.go
// registerPrefix(token.LBRACKET, p.parseArrayLiteral)
func (p *Parser) parseArrayLiteral() ast.Expression {
    array := &ast.ArrayLiteral{Token: p.curToken}
    array.Elements = p.parseExpressionList(token.RBRACKET)
    return array
}

// Generaliza parseCallArguments: sirve para ) y ]
func (p *Parser) parseExpressionList(end token.TokenType) []ast.Expression {
    list := []ast.Expression{}

    if p.peekTokenIs(end) {
        p.nextToken()
        return list
    }
    p.nextToken()
    list = append(list, p.parseExpression(LOWEST))

    for p.peekTokenIs(token.COMMA) {
        p.nextToken()
        p.nextToken()
        list = append(list, p.parseExpression(LOWEST))
    }

    if !p.expectPeek(end) {
        return nil
    }
    return list
}
parser/parser.go
const (
    _ int = iota
    LOWEST
    // [...]
    CALL  // myFunction(X)
    INDEX // array[index]
)

var precedences = map[token.TokenType]int{
    // [...]
    token.LPAREN:   CALL,
    token.LBRACKET: INDEX,
}

// registerInfix(token.LBRACKET, p.parseIndexExpression)
func (p *Parser) parseIndexExpression(left ast.Expression) ast.Expression {
    exp := &ast.IndexExpression{Token: p.curToken, Left: left}

    p.nextToken()
    exp.Index = p.parseExpression(LOWEST)

    if !p.expectPeek(token.RBRACKET) {
        return nil
    }
    return exp
}
[ tiene doble vida, igual que (: prefijo para literales [1, 2] e infijo para índices arr[0]. Con INDEX como la precedencia más alta, a * [1, 2, 3, 4][b * c] * d se parsea como ((a * ([1, 2, 3, 4][(b * c)])) * d).

5. Arrays: evaluación y funciones integradas

evaluator/evaluator.go
case *ast.ArrayLiteral:
    elements := evalExpressions(node.Elements, env)
    if len(elements) == 1 && isError(elements[0]) {
        return elements[0]
    }
    return &object.Array{Elements: elements}

func evalIndexExpression(left, index object.Object) object.Object {
    switch {
    case left.Type() == object.ARRAY_OBJ && index.Type() == object.INTEGER_OBJ:
        return evalArrayIndexExpression(left, index)
    case left.Type() == object.HASH_OBJ:
        return evalHashIndexExpression(left, index)
    default:
        return newError("index operator not supported: %s", left.Type())
    }
}

func evalArrayIndexExpression(array, index object.Object) object.Object {
    arrayObject := array.(*object.Array)
    idx := index.(*object.Integer).Value
    max := int64(len(arrayObject.Elements) - 1)

    if idx < 0 || idx > max {
        return NULL
    }
    return arrayObject.Elements[idx]
}
evaluator/builtins.go — push
"push": &object.Builtin{
    Fn: func(args ...object.Object) object.Object {
        if len(args) != 2 {
            return newError("wrong number of arguments. got=%d, want=2",
                len(args))
        }
        if args[0].Type() != object.ARRAY_OBJ {
            return newError("argument to `push` must be ARRAY, got %s",
                args[0].Type())
        }
        arr := args[0].(*object.Array)
        length := len(arr.Elements)

        newElements := make([]object.Object, length+1, length+1)
        copy(newElements, arr.Elements)
        newElements[length] = args[1]

        return &object.Array{Elements: newElements}
    },
},

Arrays inmutables: rest y push nunca modifican el array original; devuelven uno nuevo. Esto facilita razonar sobre programas funcionales.

6. Hashes: el problema de la llave

Queremos {"name": "Monkey"}["name"]. La tentación es usar map[object.Object]object.Object en Go… pero no funciona:

El problema
name1 := &object.String{Value: "name"}
monkey := &object.String{Value: "Monkey"}

pairs := map[object.Object]object.Object{}
pairs[name1] = monkey

name2 := &object.String{Value: "name"}
pairs[name2] // => nil ¡Punteros distintos!
La solución: object/object.go
type HashKey struct {
    Type  ObjectType
    Value uint64
}

func (b *Boolean) HashKey() HashKey {
    var value uint64
    if b.Value {
        value = 1
    } else {
        value = 0
    }
    return HashKey{Type: b.Type(), Value: value}
}

func (i *Integer) HashKey() HashKey {
    return HashKey{Type: i.Type(), Value: uint64(i.Value)}
}

func (s *String) HashKey() HashKey {
    h := fnv.New64a()
    h.Write([]byte(s.Value))
    return HashKey{Type: s.Type(), Value: h.Sum64()}
}
object/object.go
type HashPair struct {
    Key   Object
    Value Object
}

type Hash struct {
    Pairs map[HashKey]HashPair
}

type Hashable interface {
    HashKey() HashKey
}

¿Por qué funciona?

  • HashKey es un struct de valores: Go lo compara campo a campo, no por puntero.
  • Incluir Type evita que 1 y true choquen (ambos tendrían Value: 1).
  • HashPair guarda la llave original para poder imprimirla en Inspect().
  • Solo Integer, Boolean y String implementan Hashable.

7. Hashes: parsing y evaluación

parser/parser.go — registerPrefix(token.LBRACE, ...)
func (p *Parser) parseHashLiteral() ast.Expression {
    hash := &ast.HashLiteral{Token: p.curToken}
    hash.Pairs = make(map[ast.Expression]ast.Expression)

    for !p.peekTokenIs(token.RBRACE) {
        p.nextToken()
        key := p.parseExpression(LOWEST)

        if !p.expectPeek(token.COLON) {
            return nil
        }

        p.nextToken()
        value := p.parseExpression(LOWEST)

        hash.Pairs[key] = value

        if !p.peekTokenIs(token.RBRACE) && !p.expectPeek(token.COMMA) {
            return nil
        }
    }

    if !p.expectPeek(token.RBRACE) {
        return nil
    }
    return hash
}
evaluator/evaluator.go
func evalHashLiteral(node *ast.HashLiteral, env *object.Environment) object.Object {
    pairs := make(map[object.HashKey]object.HashPair)

    for keyNode, valueNode := range node.Pairs {
        key := Eval(keyNode, env)
        if isError(key) {
            return key
        }
        hashKey, ok := key.(object.Hashable)
        if !ok {
            return newError("unusable as hash key: %s", key.Type())
        }
        value := Eval(valueNode, env)
        if isError(value) {
            return value
        }
        hashed := hashKey.HashKey()
        pairs[hashed] = object.HashPair{Key: key, Value: value}
    }
    return &object.Hash{Pairs: pairs}
}

func evalHashIndexExpression(hash, index object.Object) object.Object {
    hashObject := hash.(*object.Hash)

    key, ok := index.(object.Hashable)
    if !ok {
        return newError("unusable as hash key: %s", index.Type())
    }
    pair, ok := hashObject.Pairs[key.HashKey()]
    if !ok {
        return NULL
    }
    return pair.Value
}
Llaves como expresiones: en {"one": 10 - 9, "thr" + "ee": 6 / 2, 4: 4, true: 5} las llaves se evalúan antes de usarse. La type assertion key.(object.Hashable) decide si el objeto puede ser llave.

8. The Grand Finale: map y reduce en Monkey

Con lo que construimos podemos escribir funciones de orden superior en Monkey mismo, sin tocar Go:

map.mk
let map = fn(arr, f) {
  let iter = fn(arr, accumulated) {
    if (len(arr) == 0) {
      accumulated
    } else {
      iter(rest(arr), push(accumulated, f(first(arr))));
    }
  };
  iter(arr, []);
};

let a = [1, 2, 3, 4];
let double = fn(x) { x * 2 };
map(a, double);
// => [2, 4, 6, 8]
reduce.mk
let reduce = fn(arr, initial, f) {
  let iter = fn(arr, result) {
    if (len(arr) == 0) {
      result
    } else {
      iter(rest(arr), f(result, first(arr)));
    }
  };
  iter(arr, initial);
};

let sum = fn(arr) {
  reduce(arr, 0, fn(initial, el) { initial + el });
};
sum([1, 2, 3, 4, 5]);
// => 15

11. Ideas clave del curso

            

                
Lexer
Caracteres → tokens con dos punteros y un peekChar.

                
Parser
Tokens → AST con descenso recursivo + Pratt para expresiones.

                
Evaluador
AST → objetos con un type switch recursivo.

                
Entornos
Mapas encadenados que dan alcance léxico y closures.

                
Extensión
Cada característica nueva recorre token → lexer → parser → objeto → eval.

                
TDD
Cada línea del intérprete fue guiada por una prueba que primero falló.
