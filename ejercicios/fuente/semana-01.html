func clasificar(ch byte) string {
    switch {
    case ch >= '0' && ch <= '9':
        return "DIGITO"
    case ch == '_' || (ch >= 'a' && ch <= 'z'):
        return "LETRA"
    case ch == ' ' || ch == '\n':
        return "ESPACIO"
    default:

    }

Contador de piezas léxicas

16 pts

Escribe contar(input string) map[string]int que recorra el string con índices (no uses strings.Fields) y cuente PALABRA, NUMERO y SIMBOLO (cualquier carácter que no sea letra, dígito ni espacio).

contar("let x = 10;")        // map[NUMERO:1 PALABRA:2 SIMBOLO:2]
contar("add(5, 10)")         // map[NUMERO:2 PALABRA:1 SIMBOLO:3]
