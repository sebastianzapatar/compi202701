Completa lo que retorna describir para cada entrada: (10 pts)
func describir(v interface{}) string {
    switch x := v.(type) {
    case int:
        return fmt.Sprintf("int %d", x*2)
    case string:
        return "string de " + strconv.Itoa(len(x))
    case nil:
        return "nil"
    default:
        return fmt.Sprintf("%T", x)
    }
}
