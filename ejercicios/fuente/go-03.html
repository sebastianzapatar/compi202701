type BaseNode struct{ Literal string }

func (b BaseNode) TokenLiteral() string { return b.Literal }
func (b BaseNode) Kind() string         { return "genérico" }
func (b BaseNode) Describe() string     { return "nodo " + b.Kind() }

type Integer struct {
    BaseNode
    Value int64
}

func (i Integer) Kind() string { return "entero" }

n := Integer{BaseNode: BaseNode{Literal: "5"}, Value: 5}
fmt.Println(n.TokenLiteral())
fmt.Println(n.Kind())
fmt.Println(n.BaseNode.Kind())
fmt.Println(n.Describe())
